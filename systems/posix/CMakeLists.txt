cmake_minimum_required(VERSION 3.20)
project(aster_host_test C CXX ASM)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|amd64|AMD64)$")
  message(FATAL_ERROR "host-posix release currently supports Linux x86_64")
endif()
set(WAMR_ROOT_DIR ${CMAKE_CURRENT_SOURCE_DIR}/../../modules/wamr)
set(WAMR_BUILD_PLATFORM "linux")
set(WAMR_BUILD_TARGET "X86_64")
set(WAMR_BUILD_INTERP 1)
set(WAMR_BUILD_FAST_INTERP 1)
set(WAMR_BUILD_AOT 1)
set(WAMR_BUILD_LIBC_BUILTIN 1)
set(WAMR_BUILD_LIBC_WASI 0)
set(WAMR_BUILD_SIMD 0)
include(${WAMR_ROOT_DIR}/build-scripts/runtime_lib.cmake)
add_library(aster_vm STATIC ${WAMR_RUNTIME_LIB_SOURCE})
target_link_libraries(aster_vm PUBLIC m dl pthread)
add_executable(aster-host main.c)
target_include_directories(aster-host PRIVATE ../../contract/include ${WAMR_ROOT_DIR}/core/iwasm/include)
target_link_libraries(aster-host PRIVATE aster_vm)
set_target_properties(aster-host PROPERTIES RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)
