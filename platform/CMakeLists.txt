# Port discovery: one directory, one Kconfig, one private build recipe.
file(GLOB ports RELATIVE ${CMAKE_CURRENT_SOURCE_DIR} ${CMAKE_CURRENT_SOURCE_DIR}/*/Kconfig)
set(selected 0)
foreach(config ${ports})
  get_filename_component(port ${config} DIRECTORY)
  string(TOUPPER ${port} symbol)
  if(CONFIG_ASTER_PLATFORM_${symbol})
    math(EXPR selected "${selected} + 1")
    add_subdirectory(${port})
  endif()
endforeach()
if(NOT selected EQUAL 1)
  message(FATAL_ERROR "Exactly one Aster hardware port must be selected; found ${selected}")
endif()
